# Supabase Schema for OnetCli

This document contains the minimal Supabase SQL schema required by the current OnetCli codebase.

## What this covers

- Authentication via Supabase Auth
- User config storage (`user_configs`)
- Subscription / Pro entitlement (`user_subscriptions`)
- Model list (`model_list`)
- Unified sync storage (`sync_data`)
- Team sharing (`teams`, `team_members`)
- RPC helper used by team invitation-by-email

## Important notes

- Supabase Auth tables are managed by Supabase itself; do not create `auth.users` manually.
- The app expects the `public` schema tables below to exist.
- `user_configs.user_id` and `user_subscriptions.user_id` are treated as one-row-per-user tables.
- `sync_data` is the main cloud sync table for both connections and workspaces.

## SQL

```sql
-- Enable UUID support if your project needs it for default keys.
create extension if not exists pgcrypto;

-- -----------------------------------------------------------------------------
-- User config storage
-- -----------------------------------------------------------------------------
create table if not exists public.user_configs (
  id bigint generated by default as identity primary key,
  user_id uuid not null,
  key_verification text not null,
  key_version integer not null default 1,
  updated_at timestamptz not null default now(),
  constraint user_configs_user_id_key unique (user_id)
);

create index if not exists user_configs_user_id_idx on public.user_configs (user_id);

-- -----------------------------------------------------------------------------
-- Subscription / Pro entitlement
-- -----------------------------------------------------------------------------
create table if not exists public.user_subscriptions (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null,
  plan text not null default 'free',
  status text not null default 'active',
  expires_at timestamptz null,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  constraint user_subscriptions_user_id_key unique (user_id),
  constraint user_subscriptions_plan_check check (plan in ('free', 'pro')),
  constraint user_subscriptions_status_check check (status in ('active', 'cancelled', 'expired'))
);

create index if not exists user_subscriptions_user_id_idx on public.user_subscriptions (user_id);

-- -----------------------------------------------------------------------------
-- Model list used by AI/chat features
-- -----------------------------------------------------------------------------
create table if not exists public.model_list (
  id uuid primary key default gen_random_uuid(),
  model text not null,
  enabled boolean not null default true,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  constraint model_list_model_key unique (model)
);

create index if not exists model_list_enabled_idx on public.model_list (enabled);
create index if not exists model_list_created_at_idx on public.model_list (created_at desc);

-- -----------------------------------------------------------------------------
-- Unified sync storage for connections/workspaces
-- -----------------------------------------------------------------------------
create table if not exists public.sync_data (
  id uuid primary key default gen_random_uuid(),
  owner_id uuid not null,
  team_id uuid null,
  data_type text not null,
  encrypted_data text not null,
  key_version integer not null default 1,
  checksum text not null default '',
  version integer not null default 1,
  updated_at timestamptz not null default now(),
  created_at timestamptz not null default now(),
  deleted_at timestamptz null,
  constraint sync_data_data_type_check check (data_type in ('connection', 'workspace'))
);

create index if not exists sync_data_owner_id_idx on public.sync_data (owner_id);
create index if not exists sync_data_team_id_idx on public.sync_data (team_id);
create index if not exists sync_data_data_type_idx on public.sync_data (data_type);
create index if not exists sync_data_updated_at_idx on public.sync_data (updated_at desc);
create index if not exists sync_data_deleted_at_idx on public.sync_data (deleted_at);

-- -----------------------------------------------------------------------------
-- Teams
-- -----------------------------------------------------------------------------
create table if not exists public.teams (
  id uuid primary key default gen_random_uuid(),
  name text not null,
  owner_id uuid not null,
  description text null,
  key_verification text null,
  key_version integer not null default 1,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create index if not exists teams_owner_id_idx on public.teams (owner_id);
create index if not exists teams_updated_at_idx on public.teams (updated_at desc);

-- -----------------------------------------------------------------------------
-- Team members
-- -----------------------------------------------------------------------------
create table if not exists public.team_members (
  id uuid primary key default gen_random_uuid(),
  team_id uuid not null references public.teams (id) on delete cascade,
  user_id uuid not null,
  role text not null default 'member',
  joined_at timestamptz not null default now(),
  constraint team_members_role_check check (role in ('owner', 'member')),
  constraint team_members_team_user_key unique (team_id, user_id)
);

create index if not exists team_members_team_id_idx on public.team_members (team_id);
create index if not exists team_members_user_id_idx on public.team_members (user_id);
create index if not exists team_members_joined_at_idx on public.team_members (joined_at asc);

-- -----------------------------------------------------------------------------
-- RPC: add team member by email
-- -----------------------------------------------------------------------------
create or replace function public.add_team_member_by_email(
  p_team_id uuid,
  p_email text
)
returns public.team_members
language plpgsql
security definer
set search_path = public
as $$
declare
  v_user_id uuid;
  v_team_owner uuid;
  v_member public.team_members;
begin
  select id into v_user_id
  from auth.users
  where lower(email) = lower(p_email)
  limit 1;

  if v_user_id is null then
    raise exception 'User not found for email %', p_email;
  end if;

  select owner_id into v_team_owner
  from public.teams
  where id = p_team_id;

  if v_team_owner is null then
    raise exception 'Team not found %', p_team_id;
  end if;

  insert into public.team_members (team_id, user_id, role)
  values (p_team_id, v_user_id, 'member')
  on conflict (team_id, user_id) do update
    set role = excluded.role
  returning * into v_member;

  return v_member;
end;
$$;
```

## Recommended RLS policy starter

If you enable Row Level Security, use this non-recursive starter first. It avoids the `team_members` self-reference loop that causes `42P17 infinite recursion` in Supabase/Postgres.

```sql
alter table public.user_configs enable row level security;
alter table public.user_subscriptions enable row level security;
alter table public.model_list enable row level security;
alter table public.sync_data enable row level security;
alter table public.teams enable row level security;
alter table public.team_members enable row level security;

-- user_configs
create policy "user_configs_select_own"
on public.user_configs for select
using (auth.uid() = user_id);

create policy "user_configs_upsert_own"
on public.user_configs for insert
with check (auth.uid() = user_id);

create policy "user_configs_update_own"
on public.user_configs for update
using (auth.uid() = user_id)
with check (auth.uid() = user_id);

-- user_subscriptions
create policy "user_subscriptions_select_own"
on public.user_subscriptions for select
using (auth.uid() = user_id);

-- model_list: read-only for authenticated users
create policy "model_list_select_authenticated"
on public.model_list for select
using (auth.role() = 'authenticated');

-- teams / team_members / sync_data
-- Keep teams readable by owner only to avoid policy recursion.
create policy "teams_owner_select"
on public.teams for select
using (auth.uid() = owner_id);

create policy "teams_owner_insert"
on public.teams for insert
with check (auth.uid() = owner_id);

create policy "teams_owner_update"
on public.teams for update
using (auth.uid() = owner_id)
with check (auth.uid() = owner_id);

create policy "teams_owner_delete"
on public.teams for delete
using (auth.uid() = owner_id);

create policy "team_members_self_or_team_owner_select"
on public.team_members for select
using (
  auth.uid() = user_id
  or exists (
    select 1 from public.teams t
    where t.id = team_members.team_id and t.owner_id = auth.uid()
  )
);

create policy "team_members_team_owner_write"
on public.team_members for all
using (
  exists (
    select 1 from public.teams t
    where t.id = team_members.team_id and t.owner_id = auth.uid()
  )
)
with check (
  exists (
    select 1 from public.teams t
    where t.id = team_members.team_id and t.owner_id = auth.uid()
  )
);

create policy "sync_data_owner_or_team_member_select"
on public.sync_data for select
using (
  auth.uid() = owner_id
  or (
    team_id is not null
    and exists (
      select 1 from public.team_members tm
      where tm.team_id = sync_data.team_id and tm.user_id = auth.uid()
    )
  )
);

create policy "sync_data_owner_or_team_member_write"
on public.sync_data for all
using (
  auth.uid() = owner_id
  or (
    team_id is not null
    and exists (
      select 1 from public.team_members tm
      where tm.team_id = sync_data.team_id and tm.user_id = auth.uid() and tm.role = 'owner'
    )
  )
)
with check (
  auth.uid() = owner_id
  or (
    team_id is not null
    and exists (
      select 1 from public.team_members tm
      where tm.team_id = sync_data.team_id and tm.user_id = auth.uid() and tm.role = 'owner'
    )
  )
);
```

## Minimal bootstrapping order

1. Enable Supabase Auth email login.
2. Run the SQL above.
3. Seed `user_subscriptions` with `plan = 'pro'` for any user that should unlock Pro features.
4. Create at least one `teams` row only if you need shared/team sync.
